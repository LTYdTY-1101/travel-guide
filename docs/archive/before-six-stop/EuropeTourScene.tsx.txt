import {useMemo} from 'react';
import type React from 'react';
import {AbsoluteFill,Easing,Img,Interactive,interpolate,staticFile,useCurrentFrame,useVideoConfig} from 'remotion';
import {Audio} from '@remotion/media';
import 'maplibre-gl/dist/maplibre-gl.css';
import {buildTour,evaluateStep,type Stop,type TourProps} from './tour-data';
import {smooth,clamp} from '../flight/flight-path';
import {useTourMap} from './use-tour-map';

const mapOpeningStyle=(seconds:number):React.CSSProperties=>{
 const interp=(keys:number[],values:number[])=>interpolate(seconds,keys,values,{extrapolateLeft:'clamp',extrapolateRight:'clamp',easing:Easing.bezier(.16,1,.3,1)});
 if(seconds>=8)return {position:'absolute',inset:0,width:1080,height:1920};
 return {position:'absolute',left:interp([3.8,4.35,6.8,7.8],[90,105,105,0]),top:interp([3.8,4.35,6.8,7.8],[685,240,240,0]),width:1080,height:1920,transformOrigin:'0 0',
  transform:`perspective(1450px) rotateY(${interp([3.8,4.35],[-88,0])}deg) rotateX(${interp([4.35,4.9,6.8,7.8],[0,25,25,0])}deg) scale(${interp([3.8,4.35,6.8,7.8],[900,870,870,1080])/1080},${interp([3.8,4.35,6.8,7.8],[234,1440,1440,1920])/1920})`,
  opacity:interp([3.75,3.95],[0,1]),borderRadius:44,overflow:'hidden',boxShadow:'0 45px 110px rgba(56,62,104,.27)'};
};

const Model:React.FC<{stop:Stop;x:number;y:number;scale:number;reveal:number;small?:boolean}>=({stop,x,y,scale,reveal,small=false})=>(
 <Interactive.Div name={`景观 · ${stop.name}`} style={{position:'absolute',left:x-stop.size*scale/2+stop.offsetX,top:y-stop.size*scale*.92+stop.offsetY,width:stop.size*scale,height:stop.size*scale*.92,opacity:reveal,transformOrigin:'50% 100%',transform:`perspective(1600px) rotateX(${-70*(1-reveal)}deg)`,filter:small?'drop-shadow(0 8px 8px rgba(30,40,40,.16))':'drop-shadow(0 22px 18px rgba(30,40,40,.22))'}}>
  <Img name={`建筑素材 · ${stop.name}`} src={staticFile(stop.image)} style={{width:'100%',height:'100%',objectFit:'contain',objectPosition:'center bottom'}} />
 </Interactive.Div>
);

export const EuropeTourScene:React.FC<TourProps>=(props)=>{
 const frame=useCurrentFrame();const {fps,width,height}=useVideoConfig();const isFull=props.section==='full';const opening=isFull?8:0;
 const {stops,legs,section}=props;
 const tour=useMemo(()=>buildTour({stops,legs,section}),[stops,legs,section]);
 const seconds=Math.max(0,Math.min(tour.seconds-.00001,frame/fps-opening));
 const step=tour.steps.find(s=>seconds<s.end)??tour.steps[tour.steps.length-1];
 const state=useMemo(()=>evaluateStep(step,seconds),[step,seconds]);
 const {container,projection:p}=useTourMap(state,props,frame,width,height);
 const closing=isFull?clamp((frame/fps-opening-tour.seconds)/.8):0;
 const openingPlane=isFull?smooth((frame/fps-5.3)/.4):1;
 const flight=step.kind==='flight',tram=step.kind==='tram',hold=step.kind==='hold';
 const vehicleSize=flight?props.planeSize*(1+.12*state.altitude):props.tramSize;
 const vehicleSrc=staticFile(flight?'plane-c4d.png':'tram-c4d.png');
 const vehicleAngle=p.angle-(tram?props.tramArtBearing:0);
 const title=flight?`${step.from.city} → ${step.to.city}`:tram?`有轨电车 · 前往${step.to.name.replace(/ [A-Z]+$/,'')}`:`D${step.day} · ${step.to.city}`;
 const dateStop=step.to.day===step.day?step.to:step.from.day===step.day?step.from:props.stops.find(s=>s.day===step.day);
 return <>
  <div data-tour-step={step.id} data-tour-kind={step.kind} data-tour-frame={frame} data-tour-ready={p.ready} style={{...(isFull?mapOpeningStyle(frame/fps):{position:'absolute',inset:0,width,height}),zIndex:2}}>
   <div ref={container} style={{position:'absolute',inset:0,width,height}} />
   <div style={{position:'absolute',inset:0,backgroundColor:'#152538',opacity:state.dim}} />
   {p.ready&&<>
    {tram&&step.from.image&&state.backdropScale>.002&&<Model stop={step.from} x={p.from.x} y={p.from.y} scale={state.backdropScale} reveal={1} small />}
    {hold&&step.to.image&&<Model stop={step.to} x={p.to.x} y={p.to.y} scale={((.86+.14*state.reveal)*(1-.75*smooth((state.local-step.seconds+.65)/.65)))} reveal={state.reveal} />}
    {!hold&&<>
     <Img name="交通工具 · 接触投影" src={vehicleSrc} style={{position:'absolute',left:p.x-vehicleSize/2+(flight?8+state.altitude*45:3),top:p.y-vehicleSize/2+(flight?10+state.altitude*58:6),width:vehicleSize,height:vehicleSize,objectFit:'contain',rotate:`${vehicleAngle}deg`,filter:`brightness(0) blur(${flight?2+state.altitude*13:3}px)`,opacity:state.opacity*(flight?.3-.21*state.altitude:.22)*openingPlane}} />
     <Img name={flight?'交通工具 · 飞机':'交通工具 · 有轨电车'} src={vehicleSrc} style={{position:'absolute',left:p.x-vehicleSize/2,top:p.y-vehicleSize/2,width:vehicleSize,height:vehicleSize,objectFit:'contain',rotate:`${vehicleAngle}deg`,opacity:state.opacity*openingPlane}} />
    </>}
    {!flight&&<Interactive.Div name="目的地 · 白色圆角名称牌" style={{position:'absolute',left:p.to.x,top:p.to.y+30,translate:'-50% 0',padding:'16px 24px',borderRadius:22,backgroundColor:'rgba(255,255,255,.96)',boxShadow:'0 12px 35px rgba(38,49,61,.17)',fontFamily:'PingFang SC, Arial, sans-serif',fontSize:32,fontWeight:650,color:'#25394b',whiteSpace:'nowrap',opacity:hold?smooth((state.local-step.to.revealDelay-.5)/.4)*(1-smooth((state.local-step.seconds+.5)/.5)):1}}>{step.to.name}</Interactive.Div>}
   </>}
   {(!isFull||frame>=opening*fps)&&<>
    <Interactive.Div name="交通与地点 · 信息条" style={{position:'absolute',left:48,top:66,maxWidth:980,padding:'17px 24px',backgroundColor:'rgba(255,255,255,.94)',borderRadius:21,boxShadow:'0 10px 32px rgba(40,60,60,.08)',fontFamily:'PingFang SC, Arial, sans-serif',fontSize:30,fontWeight:650,color:'#33485a'}}>{title}</Interactive.Div>
    <Interactive.Div name="行程 · 日期" style={{position:'absolute',left:52,top:150,padding:'10px 18px',borderRadius:16,color:'#5c7080',backgroundColor:'rgba(255,255,255,.9)',fontFamily:'PingFang SC, Arial, sans-serif',fontSize:26}}>{`D${step.day} · ${dateStop?.date??''}`}{tram?' · 市内移动示意':''}</Interactive.Div>
   </>}
   <div style={{position:'absolute',bottom:16,right:18,padding:'8px 12px',backgroundColor:'rgba(255,255,255,.94)',color:'#59646c',fontFamily:'Arial, sans-serif',fontSize:18}}>OpenFreeMap · © OpenMapTiles · Data © OpenStreetMap contributors</div>
  </div>
  {closing>0&&<AbsoluteFill style={{zIndex:4,backgroundColor:`rgba(236,242,247,${closing*.97})`,opacity:closing,padding:64,fontFamily:'PingFang SC, Arial, sans-serif',color:'#25394b'}}>
   <Interactive.Div name="终章 · 回到上海" style={{fontSize:70,fontWeight:800,marginTop:38}}>七天，十四次心动。</Interactive.Div>
   <Interactive.Div name="终章 · 环线完成" style={{fontSize:30,marginTop:22,color:'#687e8e'}}>巴黎 · 巴塞罗那 · 罗马 / 梵蒂冈 → 上海</Interactive.Div>
   <div style={{display:'grid',gridTemplateColumns:'repeat(3, 1fr)',gap:18,marginTop:52}}>
    {props.stops.filter(s=>!s.airport).map(s=><div key={s.id} style={{height:245,borderRadius:24,backgroundColor:'rgba(255,255,255,.88)',padding:12,textAlign:'center'}}><Img src={staticFile(s.image)} style={{width:'100%',height:185,objectFit:'contain'}}/><div style={{fontSize:24,fontWeight:600}}>{`D${s.day} · ${s.name}`}</div></div>)}
   </div>
   <Interactive.Div name="终章 · 再次出发" style={{marginTop:42,fontSize:38,fontWeight:650}}>已回到上海。下一次，再一起出发。</Interactive.Div>
  </AbsoluteFill>}
  <Audio name="音乐 · 全程循环与片尾淡出" src={staticFile('audio/bgm-120bpm.wav')} loop volume={props.musicVolume*smooth(frame/(fps*.5))*(1-smooth((frame/fps-(opening+tour.seconds+9))/3))} premountFor={fps}/>
  {tour.steps.filter(s=>s.kind==='hold').map(s=><Audio key={s.id} name={`音效 · ${s.to.name} 登场`} src={staticFile('audio/pop.wav')} from={Math.round((opening+s.start+s.to.revealDelay+.3)*fps)} volume={.25} premountFor={fps}/>)}
 </>;
};
