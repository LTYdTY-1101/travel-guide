import {z} from 'zod/v3';
export const tourSchema=z.object({
  section:z.enum(['full','paris','barcelona','rome','return']).describe('预览分段'),
  playbackSpeed:z.number().min(.25).max(4).describe('整体播放速度（1 为原速，3 为三倍速）'),
  planeSize:z.number().min(70).max(320).describe('飞机尺寸'),
  tramSize:z.number().min(60).max(240).describe('电车尺寸'),
  tramArtBearing:z.number().describe('电车素材原始车头方向'),
  flightColor:z.string().describe('飞行航迹颜色'),groundColor:z.string().describe('已行驶路线颜色'),
  routeWidth:z.number().min(2).max(24).describe('路线像素宽度'),cameraPitch:z.number().min(0).max(45).describe('地图俯仰'),anchorY:z.number().min(.4).max(.7).describe('交通工具画面位置'),
  showLabels:z.boolean().describe('城市地名开关'),musicVolume:z.number().min(0).max(1).describe('音乐音量'),
  stops:z.array(z.object({id:z.string(),name:z.string().describe('景点名称'),city:z.string(),day:z.number().min(1).max(7),date:z.string().describe('日期标签'),longitude:z.number(),latitude:z.number(),image:z.string().describe('建筑 PNG（public 相对路径）'),airport:z.boolean(),zoom:z.number().min(10).max(18).describe('地点缩放'),holdSeconds:z.number().min(0).max(30).describe('景点停留秒数'),size:z.number().min(150).max(1100).describe('建筑尺寸'),revealDelay:z.number().min(0).max(3).describe('建筑登场延迟'),offsetX:z.number().describe('建筑相对锚点横向偏移'),offsetY:z.number().describe('建筑相对锚点纵向偏移')})).describe('逐个地点 · 名称、日期、位置、建筑和停留时间'),
  legs:z.array(z.object({id:z.string(),kind:z.enum(['flight','tram']),from:z.string(),to:z.string(),seconds:z.number().min(1.2).max(60).describe('本交通段秒数'),day:z.number().min(1).max(7),cruiseZoom:z.number().min(2).max(10).describe('飞行巡航缩放')})).describe('逐段交通 · 飞机、电车和机场接驳时长'),
});
